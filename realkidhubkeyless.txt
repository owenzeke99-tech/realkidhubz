-- File lưu trữ Key trên Executor
local SAVE_FILE = "RealKidHub_SavedKey.txt"
local KEY_DURATION = 86400 -- 24 giờ (tính bằng giây)

-- Danh sách Key chuẩn
local ValidKeys = {
    "RealKidHubKeyKsjwowon - KalwojwbekekdnHalwoeuhejeke",
    "RealKidHubKeyKsjwowon-KalwojwbekekdnHalwoeuhejeke",
    "RealKidHubKeyKsjwowon",
    "KalwojwbekekdnHalwoeuhejeke"
}

-- Hàm làm sạch chuỗi (Lược bỏ khoảng trắng thừa ở đầu/cuối)
local function CleanString(str)
    return (str:gsub("^%s*(.-)%s*$", "%1"))
end

-- Hàm kiểm tra Key
local function IsValidKey(key)
    local cleaned = CleanString(key)
    for _, k in ipairs(ValidKeys) do
        if cleaned == k then
            return true
        end
    end
    return false
end

-- Hàm Lưu Key vào File (Lưu định dạng Chuỗi đơn giản: KEY|TIMESTAMP)
local function SaveKeyData(key)
    if writefile then
        pcall(function()
            local content = CleanString(key) .. "|" .. tostring(os.time())
            writefile(SAVE_FILE, content)
        end)
    end
end

-- Hàm Kiểm Tra Key Đã Lưu (Bỏ qua menu nếu còn hạn 24 giờ)
local function CheckSavedKey()
    if readfile and isfile and isfile(SAVE_FILE) then
        local success, content = pcall(function()
            return readfile(SAVE_FILE)
        end)
        
        if success and content then
            local savedKey, savedTime = content:match("^([^|]+)|(%d+)$")
            if savedKey and savedTime then
                local timePassed = os.time() - tonumber(savedTime)
                if timePassed < KEY_DURATION and IsValidKey(savedKey) then
                    return true
                end
            end
        end
    end
    return false
end

-- TỰ ĐỘNG BỎ QUA MENU NẾU KEY ĐÃ ĐƯỢC LƯU VÀ CÒN HẠN 24H
if CheckSavedKey() then
    loadstring(game:HttpGet(SCRIPT_URL))()
    return
end

-- Shake Effect
local function SmoothShake()
    local shakeOffsets = {10, -8, 6, -4, 2, 0}
    
    TweenService:Create(MainStroke, TweenInfo.new(0.1), {Color = ERROR_RED}):Play()
    TweenService:Create(KeyBoxStroke, TweenInfo.new(0.1), {Color = ERROR_RED}):Play()

    task.spawn(function()
        for _, offset in ipairs(shakeOffsets) do
            local targetPos = UDim2.new(0.5, -190 + offset, 0.5, -112)
            local tween = TweenService:Create(MainFrame, TweenInfo.new(0.04, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {Position = targetPos})
            tween:Play()
            tween.Completed:Wait()
        end
        
        MainFrame.Position = BasePosition
        TweenService:Create(MainStroke, TweenInfo.new(0.25), {Color = ICE_BLUE}):Play()
        TweenService:Create(KeyBoxStroke, TweenInfo.new(0.25), {Color = Color3.fromRGB(60, 80, 110)}):Play()
    end)
end
